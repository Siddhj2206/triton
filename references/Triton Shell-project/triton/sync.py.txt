#!/usr/bin/env python3
"""Keep every board on the shared Triton tokens.

Run from the project folder:  python3 triton/sync.py
1. Injects triton/tokens.css and triton/components.css into each board's <helmet><style>
   between /* triton:begin */ and /* triton:end */ (replaced on every run).
2. Rewrites the material recipes that drifted between boards (Mica, Glass, Thick Glass)
   to the canonical values, via the --t-mica / --t-glass / --t-thick variables.
Edit the CSS files, run this, republish. Boards stay self-contained, so they render on the canvas.
"""
import glob,re,os
here=os.path.dirname(os.path.abspath(__file__)); proj=os.path.dirname(here)
shared=open(os.path.join(here,'tokens.css')).read()+'\n'+open(os.path.join(here,'components.css')).read()
BLOCK='/* triton:begin */\n'+shared+'\n/* triton:end */\n'
SUBS=[
 (r'rgb\(24 27 34 / \.78\)|rgb\(17 19 24 / \.72\)','var(--t-mica)'),
 (r'(backdrop-filter:)blur\(40px\) saturate\(1\.3\)',r'\1blur(48px) saturate(1.4) brightness(.8)'),
 (r'(backdrop-filter:)blur\(48px\) saturate\(1\.45\)',r'\1blur(48px) saturate(1.4) brightness(.8)'),
 (r'(-webkit-backdrop-filter:)blur\(40px\) saturate\(1\.3\)',r'\1blur(48px) saturate(1.4) brightness(.8)'),
 (r'rgb\(40 42 47 / \.55\)','var(--t-thick)'),
 (r'rgb\(40 42 47 / \.88\)','var(--t-thick)'),
 (r'rgb\(40 42 47 / \.62\)','var(--t-glass)'),
 (r'(backdrop-filter:)blur\(48px\) saturate\(1\.5\) brightness\(\.8\)',r'\1blur(40px) saturate(1.4) brightness(.8)'),
 (r'(-webkit-backdrop-filter:)blur\(48px\) saturate\(1\.5\) brightness\(\.8\)',r'\1blur(40px) saturate(1.4) brightness(.8)'),
]
n=0
for f in sorted(glob.glob(os.path.join(proj,'*.dc.html'))):
    s=open(f).read(); o=s
    s=re.sub(r'/\* triton:begin \*/.*?/\* triton:end \*/\n','',s,flags=re.S)
    for a,b in SUBS: s=re.sub(a,b,s)
    i=s.index('<style>')+len('<style>')
    s=s[:i]+'\n'+BLOCK+s[i:]
    if s!=o: open(f,'w').write(s); n+=1
print('updated',n,'boards')
